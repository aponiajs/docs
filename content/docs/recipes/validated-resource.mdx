---
title: Validate a resource
description: Own one schema file per feature and derive both handler types and route validation from it.
---

Keep every validator for a feature in one file, and derive the handler types
from the same declarations. A field is then declared once.

```ts title="src/items/item.schema.ts"
import { z } from "zod";

export const createItemSchema = {
  body: z.object({
    name: z.string().min(2),
    quantity: z.number().int().positive(),
  }),
};

export const findItemSchema = {
  params: z.object({ id: z.string().min(1) }),
};

export type CreateItem = z.infer<(typeof createItemSchema)["body"]>;
```

```ts title="src/items/item.service.ts"
import { Injectable } from "@aponiajs/common";
import type { CreateItem } from "./item.schema.ts";

export interface StoredItem extends CreateItem {
  readonly id: string;
}

@Injectable()
export class ItemService {
  readonly #items = new Map<string, StoredItem>();

  create(item: CreateItem): StoredItem {
    const created: StoredItem = { id: crypto.randomUUID(), ...item };
    this.#items.set(created.id, created);
    return created;
  }

  findOne(id: string): StoredItem | undefined {
    return this.#items.get(id);
  }
}
```

```ts title="src/items/item.controller.ts"
import { Body, Controller, Get, Param, Post } from "@aponiajs/common";
import { createItemSchema, findItemSchema, type CreateItem } from "./item.schema.ts";
import { ItemService } from "./item.service.ts";

@Controller("items")
export class ItemController {
  constructor(private readonly itemService: ItemService) {}

  @Post("/", createItemSchema)
  create(@Body() body: CreateItem) {
    return this.itemService.create(body);
  }

  @Get(":id", findItemSchema)
  findOne(@Param("id") id: string) {
    return this.itemService.findOne(id);
  }
}
```

```ts title="src/items/item.module.ts"
import { Module } from "@aponiajs/common";
import { ItemController } from "./item.controller.ts";
import { ItemService } from "./item.service.ts";

@Module({
  controllers: [ItemController],
  providers: [ItemService],
})
export class ItemModule {}
```

## Use Elysia's `t` instead

Elysia's builder ships with the platform peer dependency, so no extra install is
needed. It is also what `aponia g resource <name> --type rest` generates:

```ts
import { t } from "elysia";
import type { Static } from "elysia";

export const createItemSchema = {
  body: t.Object({
    name: t.String({ minLength: 2 }),
    quantity: t.Integer({ minimum: 1 }),
  }),
};

export type CreateItem = Static<(typeof createItemSchema)["body"]>;
```

`t.Numeric()` is the right choice for a query or path value that must arrive as
a number, because those parts are strings on the wire.

## Test it

```ts
import { expect, test } from "bun:test";
import { AponiaFactory } from "@aponiajs/platform-elysia";
import { AppModule } from "../src/app.module.ts";

test("rejects an invalid item", async () => {
  const application = await AponiaFactory.create(AppModule, { logger: false });

  const rejected = await application.handle(
    new Request("http://localhost/items", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ name: "A", quantity: 0 }),
    }),
  );

  expect(rejected.status).toBe(422);
});
```
